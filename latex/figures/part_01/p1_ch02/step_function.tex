\definecolor{myred}{RGB}{190,60,45}

% 把每张图做成命令，保证坐标系完全一致 → y 轴必然对齐
\newcommand{\stepfig}{%
\begin{tikzpicture}[x=1.1cm, y=1.8cm, >=stealth, line width=0.8pt, font=\large, baseline=(origin)]
  \draw[->] (-3,0) -- (3,0)      node[below right] {$x$};
  \draw[->] (0,-0.6) -- (0,1.5)  node[above right] {$y$};
  \coordinate (origin) at (0,0);            % 对齐基准点
  \draw (0,1) -- ++(-0.08,0);
  \draw[myred, line width=1pt] (-3,0) -- (0,0);
  \draw[myred, line width=1pt] (0,1)  -- (3,1);
  \draw[myred, fill=white, line width=1pt] (0,0) circle (2.5pt);
  \fill[myred] (0,1) circle (2.5pt);
  \node[left] at (-0.12,1) {$1$};
  \node[below left] at (-0.06,-0.08) {$O$};
\end{tikzpicture}}

\newcommand{\derivfig}{%
\begin{tikzpicture}[x=1.1cm, y=1.8cm, >=stealth, line width=0.8pt, font=\large, baseline=(origin)]
  \draw[->] (-3,0) -- (3,0)      node[below right] {$x$};
  \draw[->] (0,-0.6) -- (0,1.5)  node[above right] {$y'$};
  \coordinate (origin) at (0,0);
  \draw[myred, line width=1pt] (-3,0) -- (3,0);
  \draw[myred, fill=white, line width=1pt] (0,0) circle (2.5pt);
  \node[below left] at (-0.06,-0.08) {$O$};
\end{tikzpicture}}
